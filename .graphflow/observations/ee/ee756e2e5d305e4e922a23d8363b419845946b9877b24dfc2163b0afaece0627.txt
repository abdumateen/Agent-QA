39: def _normalize_tags(tags: list[str]) -> list[str]:
40:     normalized: list[str] = []
41:     seen: set[str] = set()
42:     for tag in tags:
43:         cleaned = tag.strip()
44:         if not cleaned:
45:             raise ValueError("Tags must not be empty.")
46:         if "," in cleaned:
47:             raise ValueError("Tags must not contain commas.")
48:         if any(ord(character) < 32 or ord(character) == 127 for character in cleaned):
49:             raise ValueError("Tags must not contain control characters.")
50:         if cleaned not in seen:
51:             seen.add(cleaned)
52:             normalized.append(cleaned)
53:     return normalized
